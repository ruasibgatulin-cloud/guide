Dodge the Creeps — добавляем HP, урон и замедление

Этот гайд дополняет официальный туториал Godot "Dodge the Creeps". Мы добавим:

  • 100 HP у игрока (вместо смерти с одного удара)
  • Урон 10–20 за касание моба
  • Неуязвимость после получения урона
  • Бонус-звёздочку, которая замедляет мобов на 5 секунд

Содержание

  1. Часть 1. Настраиваем Mob (врага)
  2. Часть 2. Настраиваем Player (игрока)
  3. Часть 3. Создаём Pickup (звёздочку)
  4. Часть 4. Обновляем HUD
  5. Часть 5. Обновляем Main
  6. Часть 6. Проверка
  7. Если что-то сломалось
Ориентиры на экране: слева — панель "Сцена" (дерево объектов), справа — "Инспектор" (настройки), центр — игровой мир, нижний левый угол — "Файловая система".

Часть 1. Настраиваем Mob (врага)

Шаг 1. Открываем сцену Mob.
Смотри в нижний левый угол — там список файлов. Найди mob.tscn и дважды щёлкни по нему левой кнопкой мыши.
Шаг 2. Добавляем Mob в группу "mobs".
1. В левом верхнем углу в дереве сцены щёлкни на самый верхний узел — Mob.
2. Смотри вправо — там панель "Инспектор". В самом верху ряд вкладок: Инспектор | Узел | История.
3. Нажми вкладку Узел.
4. Найди строку Группы и нажми кнопку + справа от неё.
5. Введи mobs и нажми Enter.
Шаг 3. Подключаем сигнал body_entered у Mob.
1. Там же вкладка Узел.
2. Раскрой раздел Сигналы.
3. Найди в списке body_entered и дважды щёлкни по нему.
4. В поле "Receiver Method" напиши _on_body_entered.
5. Нажми кнопку Connect.
Шаг 4. Заменяем скрипт mob.gd.
1. В дереве сцены щёлкни правой кнопкой по Mob → Открыть скрипт.
2. В окне кода: Ctrl+A → Delete.
3. Вставь код ниже.
mob.gd 
extends RigidBody2D

@export var min_damage: int = 10
@export var max_damage: int = 20

var _base_linear_velocity: Vector2
var _slow_timer: float = 0.0

func _ready():
	add_to_group("mobs")
	var mob_types = Array($AnimatedSprite2D.sprite_frames.get_animation_names())
	$AnimatedSprite2D.animation = mob_types.pick_random()
	$AnimatedSprite2D.play()
	_base_linear_velocity = linear_velocity


func _physics_process(delta):
	if _slow_timer > 0.0:
		_slow_timer -= delta
		if _slow_timer <= 0.0:
			linear_velocity = _base_linear_velocity


func apply_slow(factor: float, duration: float):
	_slow_timer = duration
	if _base_linear_velocity == Vector2.ZERO:
		_base_linear_velocity = linear_velocity
	linear_velocity = _base_linear_velocity * factor


func _on_body_entered(body):
	if body.is_in_group("player"):
		var dmg = randi_range(min_damage, max_damage)
		body.take_damage(dmg)


func _on_VisibilityNotifier2D_screen_exited():
	queue_free()

Нажми Ctrl+S, чтобы сохранить.

Важно! Если Godot подчеркнёт _on_VisibilityNotifier2D_screen_exited красным — проверь, что узел VisibilityNotifier2D есть в сцене Mob и его сигнал screen_exited подключён к этой функции.

Часть 2. Настраиваем Player (игрока)

Шаг 5. Открываем сцену Player.
В нижнем левом углу найди player.tscn и дважды щёлкни по нему.
Шаг 6. Добавляем Player в группу "player".
1. В дереве сцены (слева вверху) щёлкни на корень Player.
2. Справа в Инспекторе нажми вкладку Узел.
3. Найди строку Группы → нажми +.
4. Введи player и нажми Enter.
Шаг 7. Проверяем сигнал body_entered у Player.
В Инспекторе → вкладка Узел → раздел Сигналы. Если у body_entered стоит зелёная галочка — ничего не делай. Если не подключён — подключи к _on_body_entered как в шаге 3.
Шаг 8. Заменяем скрипт player.gd.
Правой кнопкой по Player → Открыть скрипт → Ctrl+A → Delete → вставь код.
player.gd 
extends Area2D

signal hit
signal health_changed(new_health)

@export var speed = 400
@export var max_health: int = 100
@export var invulnerability_time: float = 1.5

var screen_size
var health: int
var _invulnerable := false

func _ready():
	screen_size = get_viewport_rect().size
	health = max_health
	add_to_group("player")
	hide()


func _process(delta):
	var velocity = Vector2.ZERO
	if Input.is_action_pressed(&"move_right"):
		velocity.x += 1
	if Input.is_action_pressed(&"move_left"):
		velocity.x -= 1
	if Input.is_action_pressed(&"move_down"):
		velocity.y += 1
	if Input.is_action_pressed(&"move_up"):
		velocity.y -= 1

	if velocity.length() > 0:
		velocity = velocity.normalized() * speed
		$AnimatedSprite2D.play()
	else:
		$AnimatedSprite2D.stop()

	position += velocity * delta
	position = position.clamp(Vector2.ZERO, screen_size)

	if velocity.x != 0:
		$AnimatedSprite2D.animation = &"right"
		$AnimatedSprite2D.flip_v = false
		$Trail.rotation = 0
		$AnimatedSprite2D.flip_h = velocity.x < 0
	elif velocity.y != 0:
		$AnimatedSprite2D.animation = &"up"
		rotation = PI if velocity.y > 0 else 0


func start(pos):
	position = pos
	rotation = 0
	health = max_health
	health_changed.emit(health)
	_invulnerable = false
	modulate.a = 1.0
	show()
	$CollisionShape2D.disabled = false


func take_damage(amount: int):
	if _invulnerable:
		return
	health -= amount
	health_changed.emit(health)
	if health <= 0:
		_die()
	else:
		_start_invulnerability()


func _start_invulnerability():
	_invulnerable = true
	var tween = create_tween()
	tween.set_loops(6)
	tween.tween_property(self, "modulate:a", 0.3, 0.1)
	tween.tween_property(self, "modulate:a", 1.0, 0.1)
	await get_tree().create_timer(invulnerability_time).timeout
	_invulnerable = false
	modulate.a = 1.0


func _die():
	hide()
	hit.emit()
	$CollisionShape2D.set_deferred(&"disabled", true)


func _on_body_entered(_body):
	pass

Нажми Ctrl+S.

Почему _on_body_entered пустая? В официальном туториале сигнал body_entered уже подключён к этой функции. Если её удалить — Godot будет ругаться. Мы оставили её пустой, а всю работу делает Mob.

Часть 3. Создаём Pickup (звёздочку)

Шаг 9. Создаём новую сцену.
В правом верхнем углу меню: Сцена | Проект | Отладка | ... Нажми Сцена → Новая сцена.
Шаг 10. Меняем корневой узел на Area2D.
1. В левом верхнем углу в дереве сейчас стоит какой-то узел.
2. В правом верхнем углу панели "Инспектор" есть кнопка Другой узел (или иконка лупы).
3. Нажми её → в поиске напиши Area2D → выбери → Заменить.
Шаг 11. Переименовываем узел.
Правой кнопкой по Area2D → Переименовать → напиши Pickup → Enter.
Шаг 12. Добавляем картинку.
1. В левом верхнем углу над деревом есть кнопка + ("Добавить узел").
2. В поиске напиши Sprite2D → выбери → Создать.
Шаг 13. Ставим картинку.
В Инспекторе найди Texture → нажми на пустое поле <Пусто> → выбери любую картинку (например icon.svg). При желании поменяй Scale на 2 и 2.
Шаг 14. Добавляем форму столкновения.
1. Снова нажми + → напиши CollisionShape2D → Создать.
2. В Инспекторе найди Shape → нажми на пустое поле → New CircleShape2D.
3. Раскрой его и поставь Radius = 30.
Шаг 15. Сохраняем сцену.
Ctrl+S → в окошке имя файла: pickup.tscn → Сохранить.
Шаг 16. Вешаем скрипт.
Правой кнопкой по Pickup → Присоединить скрипт → Создать → Ctrl+A → Delete → вставь код.
pickup.gd 
extends Area2D

@export var slow_duration: float = 5.0
@export var slow_factor: float = 0.4

var _used := false

func _ready():
	body_entered.connect(_on_body_entered)


func _on_body_entered(body):
	if _used:
		return
	if not body.is_in_group("player"):
		return
	_used = true
	get_tree().call_group("mobs", "apply_slow", slow_factor, slow_duration)
	queue_free()

Нажми Ctrl+S.

Шаг 21. Заменяем main.gd.
Правой кнопкой по Main → Открыть скрипт → Ctrl+A → Delete → вставь код.
main.gd 
extends Node

@export var mob_scene: PackedScene
@export var pickup_scene: PackedScene
@export var pickup_spawn_interval: float = 7.0

var score


func game_over():
	$ScoreTimer.stop()
	$MobTimer.stop()
	$HUD.show_game_over()
	$Music.stop()
	$DeathSound.play()


func new_game():
	get_tree().call_group(&"mobs", &"queue_free")
	score = 0
	$Player.start($StartPosition.position)
	$Player.health_changed.connect($HUD.update_health)
	$HUD.update_health($Player.max_health)
	$StartTimer.start()
	$HUD.update_score(score)
	$HUD.show_message("Get Ready")
	$Music.play()


func _ready():
	var t = Timer.new()
	t.wait_time = pickup_spawn_interval
	t.autostart = true
	t.timeout.connect(_spawn_pickup)
	add_child(t)


func _spawn_pickup():
	if not pickup_scene:
		return
	var pickup = pickup_scene.instantiate()
	var screen = get_viewport_rect().size
	pickup.position = Vector2(
		randf_range(80, screen.x - 80),
		randf_range(80, screen.y - 80)
	)
	add_child(pickup)


func _on_MobTimer_timeout():
	var mob = mob_scene.instantiate()
	var mob_spawn_location = get_node(^"MobPath/MobSpawnLocation")
	mob_spawn_location.progress_ratio = randf()
	mob.position = mob_spawn_location.position
	var direction = mob_spawn_location.rotation + PI / 2
	direction += randf_range(-PI / 4, PI / 4)
	mob.rotation = direction
	var velocity = Vector2(randf_range(150.0, 250.0), 0.0)
	mob.linear_velocity = velocity.rotated(direction)
	add_child(mob)


func _on_ScoreTimer_timeout():
	score += 1
	$HUD.update_score(score)


func _on_StartTimer_timeout():
	$MobTimer.start()
	$ScoreTimer.start()

Нажми Ctrl+S.

Шаг 22. Привязываем pickup.tscn к Main.
1. В дереве сцены щёлкни на корень Main.
2. В Инспекторе найди строчку Pickup Scene.

Часть 6. Проверка

  1. Нажми F5 (или кнопку ▶ в правом верхнем углу).
  2. Нажми Start в игре.
  3. Управление: стрелки на клавиатуре.
  4. Через 7 секунд появится звёздочка. Подбери её — мобы замедлятся.
  5. Дай мобу тебя ударить — HP упадёт на 10–20, и ты помигаешь.
  6. Когда HP = 0 — игра закончится.
